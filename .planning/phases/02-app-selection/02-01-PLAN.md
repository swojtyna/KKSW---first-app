```swift
struct Blocklist: Codable, Equatable {
    var id: UUID
    var name: String?
    var records: [TokenRecord]
    /// Snapshot of the last picker commit. Drives (a) picker pre-seeding on re-open (SEL-04)
    /// and (b) scenePhase reconcile (SEL-05). Kept in sync with `records` by every UseCase
    /// that mutates either side (Pitfall 1 in RESEARCH.md).
    var selection: FamilyActivitySelection

    static let empty = Blocklist(
        id: UUID(),
        name: nil,
        records: [],
        selection: FamilyActivitySelection()
    )
}

struct TokenRecord: Codable, Equatable, Identifiable {
    let id: UUID                       // primary key, NEVER derived from token bytes (SEL-05)
    let kind: Kind
    var token: TokenPayload

    enum Kind: String, Codable { case application, category, webDomain }

    enum TokenPayload: Codable, Equatable, Hashable {
        case application(ApplicationToken)
        case category(ActivityCategoryToken)
        case webDomain(WebDomainToken)
    }
}
```

```swift
import Foundation
import FamilyControls

struct BlocklistSnapshot: Equatable {
    var apps: [TokenRecord]
    var categories: [TokenRecord]
    var webDomains: [TokenRecord]
    var selection: FamilyActivitySelection   // used to pre-seed the picker

    var isEmpty: Bool { apps.isEmpty && categories.isEmpty && webDomains.isEmpty }

    static let empty = BlocklistSnapshot(
        apps: [], categories: [], webDomains: [],
        selection: FamilyActivitySelection()
    )

    init(apps: [TokenRecord], categories: [TokenRecord], webDomains: [TokenRecord], selection: FamilyActivitySelection) {
        self.apps = apps
        self.categories = categories
        self.webDomains = webDomains
        self.selection = selection
    }

    init(from blocklist: Blocklist) {
        // Stable sort by record UUID: opaque, privacy-safe, rotation-independent.
        func sorted(_ r: [TokenRecord]) -> [TokenRecord] {
            r.sorted { $0.id.uuidString < $1.id.uuidString }
        }
        self.apps        = sorted(blocklist.records.filter { $0.kind == .application })
        self.categories  = sorted(blocklist.records.filter { $0.kind == .category })
        self.webDomains  = sorted(blocklist.records.filter { $0.kind == .webDomain })
        self.selection   = blocklist.selection
    }
}
```

**File 5 — `Domain/UseCases/UpdateBlocklistFromSelectionUseCase.swift`:**

```swift
import Foundation
import FamilyControls

protocol UpdateBlocklistFromSelectionUseCase: Sendable {
    func callAsFunction(newSelection: FamilyActivitySelection) async throws
}

final class UpdateBlocklistFromSelectionUseCaseImpl: UpdateBlocklistFromSelectionUseCase {
    private let repo: BlocklistRepository
    init(repo: BlocklistRepository) { self.repo = repo }

    func callAsFunction(newSelection: FamilyActivitySelection) async throws {
        var blocklist = try await repo.load()

        // Index existing records by their token payload so we can reuse UUIDs (SEL-05).
        var existingByToken: [TokenRecord.TokenPayload: TokenRecord] = [:]
        for r in blocklist.records { existingByToken[r.token] = r }

        var next: [TokenRecord] = []

        for t in newSelection.applicationTokens {
            let key: TokenRecord.TokenPayload = .application(t)
            next.append(existingByToken[key] ?? TokenRecord(id: UUID(), kind: .application, token: key))
        }
        for t in newSelection.categoryTokens {
            let key: TokenRecord.TokenPayload = .category(t)
            next.append(existingByToken[key] ?? TokenRecord(id: UUID(), kind: .category, token: key))
        }
        for t in newSelection.webDomainTokens {
            let key: TokenRecord.TokenPayload = .webDomain(t)
            next.append(existingByToken[key] ?? TokenRecord(id: UUID(), kind: .webDomain, token: key))
        }

        blocklist.records = next
        blocklist.selection = newSelection         // Pitfall 1: MUST update both sides atomically
        try await repo.save(blocklist)
    }
}
```

```swift
import Foundation

protocol RemoveTokenFromBlocklistUseCase: Sendable {
    func callAsFunction(recordId: UUID) async throws
}

final class RemoveTokenFromBlocklistUseCaseImpl: RemoveTokenFromBlocklistUseCase {
    private let repo: BlocklistRepository
    init(repo: BlocklistRepository) { self.repo = repo }

    func callAsFunction(recordId: UUID) async throws {
        var blocklist = try await repo.load()
        guard let removed = blocklist.records.first(where: { $0.id == recordId }) else { return }
        blocklist.records.removeAll { $0.id == recordId }

        // Pitfall 1: keep Blocklist.selection in sync so the picker doesn't re-seed deleted items.
        switch removed.token {
```

```swift
import Foundation
import FamilyControls

protocol ReconcileBlocklistUseCase: Sendable {
    func callAsFunction() async throws
}

final class ReconcileBlocklistUseCaseImpl: ReconcileBlocklistUseCase {
    private let repo: BlocklistRepository
    init(repo: BlocklistRepository) { self.repo = repo }

    func callAsFunction() async throws {
        var blocklist = try await repo.load()
        let fresh = blocklist.selection

        var didChange = false

        // Forward pass: rewrite each record.token under its stable UUID
        // with the fresh value from `selection` (handles silent rotation).
        for i in blocklist.records.indices {
            let record = blocklist.records[i]
            switch record.token {
            case .application(let old):
                if let match = fresh.applicationTokens.first(where: { $0 == old }) {
                    let newPayload: TokenRecord.TokenPayload = .application(match)
                    if blocklist.records[i].token != newPayload {
                        blocklist.records[i].token = newPayload
                        didChange = true
                    }
                }
            case .category(let old):
                if let match = fresh.categoryTokens.first(where: { $0 == old }) {
                    let newPayload: TokenRecord.TokenPayload = .category(match)
                    if blocklist.records[i].token != newPayload {
                        blocklist.records[i].token = newPayload
                        didChange = true
                    }
                }
            case .webDomain(let old):
                if let match = fresh.webDomainTokens.first(where: { $0 == old }) {
                    let newPayload: TokenRecord.TokenPayload = .webDomain(match)
                    if blocklist.records[i].token != newPayload {
                        blocklist.records[i].token = newPayload
                        didChange = true
                    }
                }
            }
        }

        // Reverse pass: any fresh token NOT represented as a record gets a fresh UUID.
        // Covers crash-mid-write or mixed-source edge cases.
        let existing: Set<TokenRecord.TokenPayload> = Set(blocklist.records.map(\.token))

        for t in fresh.applicationTokens where !existing.contains(.application(t)) {
            blocklist.records.append(TokenRecord(id: UUID(), kind: .application, token: .application(t)))
            didChange = true
        }
        for t in fresh.categoryTokens where !existing.contains(.category(t)) {
            blocklist.records.append(TokenRecord(id: UUID(), kind: .category, token: .category(t)))
            didChange = true
        }
        for t in fresh.webDomainTokens where !existing.contains(.webDomain(t)) {
            blocklist.records.append(TokenRecord(id: UUID(), kind: .webDomain, token: .webDomain(t)))
            didChange = true
        }

        if didChange { try await repo.save(blocklist) }
    }
}
```

```swift
                                    category: "BlocklistRepository")
    static let defaultAppGroupID = "group.com.kksw.DeluluDetox"
    static let fileName = "blocklists.v1.json"
    private static let darwinChangeName = "com.kksw.DeluluDetox.blocklists.changed" as CFString

    private let containerURL: URL

    /// Production initializer — resolves the shared App Group container.
    convenience init(appGroupID: String = BlocklistRepositoryImpl.defaultAppGroupID) throws {
        guard let url = FileManager.default
            .containerURL(forSecurityApplicationGroupIdentifier: appGroupID) else {
            throw BlocklistRepositoryError.containerUnavailable
        }
        self.init(containerURL: url)
    }

    /// Test initializer — inject a temp directory (RESEARCH.md Gap 7).
    init(containerURL: URL) {
        self.containerURL = containerURL
    }

    private var fileURL: URL {
        containerURL.appendingPathComponent(Self.fileName, isDirectory: false)
    }

    // MARK: - Load

    func load() async throws -> Blocklist {
        let url = fileURL
        let coordinator = NSFileCoordinator(filePresenter: nil)
        var coordErr: NSError?
        var result: Result<Blocklist, Error> = .success(.empty)

        coordinator.coordinate(readingItemAt: url,
                               options: .withoutChanges,
                               error: &coordErr) { coordURL in
            do {
                let data = try Data(contentsOf: coordURL)
                do {
                    let decoded = try JSONDecoder().decode(Blocklist.self, from: data)
                    result = .success(decoded)
                } catch {
                    Self.log.error("Blocklist decode failed: \(error.localizedDescription, privacy: .public)")
                    result = .failure(BlocklistRepositoryError.decode(error))
                }
            } catch CocoaError.fileReadNoSuchFile {
                // First launch — no file yet. Treat as empty, not an error.
                result = .success(.empty)
            } catch {
                Self.log.error("Blocklist read failed: \(error.localizedDescription, privacy: .public)")
                result = .failure(BlocklistRepositoryError.readFailed(error))
            }
        }
        if let e = coordErr { throw BlocklistRepositoryError.coordination(e) }
        return try result.get()
    }

    // MARK: - Save

    func save(_ blocklist: Blocklist) async throws {
        let url = fileURL
        let data = try JSONEncoder().encode(blocklist)  // NEVER PropertyListEncoder (forum 721973)

        // Ensure the directory exists (critical for the temp-dir test path).
        try FileManager.default.createDirectory(at: url.deletingLastPathComponent(),
                                                withIntermediateDirectories: true)

        let coordinator = NSFileCoordinator(filePresenter: nil)
        var coordErr: NSError?
        var writeErr: Error?

        coordinator.coordinate(writingItemAt: url,
                               options: .forReplacing,
                               error: &coordErr) { coordURL in
            do {
                try data.write(to: coordURL,
                               options: [.atomic, .completeFileProtectionUntilFirstUserAuthentication])
            } catch { writeErr = error }
        }
        if let e = coordErr { throw BlocklistRepositoryError.coordination(e) }
        if let e = writeErr { throw BlocklistRepositoryError.writeFailed(e) }

        Self.log.info("Blocklist saved, records=\(blocklist.records.count, privacy: .public)")
        postChangeNotification()
    }

    // MARK: - Changes stream

    func changes() -> AsyncStream<Void> {
        AsyncStream { continuation in
            let center = CFNotificationCenterGetDarwinNotifyCenter()
            let name = CFNotificationName(Self.darwinChangeName)
            let observer = Unmanaged.passUnretained(self).toOpaque()

            CFNotificationCenterAddObserver(
                center, observer,
                { _, observer, _, _, _ in
                    guard let observer = observer else { return }
                    // Hop to task via a main-actor-independent notification bridge in consumer; we emit Void.
                    _ = observer
                },
                Self.darwinChangeName, nil,
                .deliverImmediately
            )
            continuation.onTermination = { _ in
                CFNotificationCenterRemoveObserver(center, observer, name, nil)
            }
        }
    }

    private func postChangeNotification() {
        CFNotificationCenterPostNotification(
            CFNotificationCenterGetDarwinNotifyCenter(),
            CFNotificationName(Self.darwinChangeName),
            nil, nil, true
        )
    }
}
```
