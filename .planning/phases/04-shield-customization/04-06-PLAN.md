<objective>
Close SHL-03 dispatch gap by shipping a local-push-notification fallback so that tapping the shield primary button actually foregrounds DeluluDetox on the correct destination.

Purpose: Phase 4 UAT on iPhone 14 Pro (iOS 26.3.1, build `e89c7c1`) verified SHL-01 and SHL-04 pass but reported SHL-03 as Partial — the shield dismisses but never hands off to the main app. `ShieldActionDelegate` inherits from `NSObject` (not UIViewController) and exposes no `extensionContext`; the Objective-C runtime workaround is App Store Guideline 2.5.1 rejection risk and was reverted in commit `3060296`. This plan wires the only remaining acceptable path: extension schedules a local `UNNotificationRequest` with `userInfo["url"]`, main app's `UNUserNotificationCenterDelegate` translates a banner tap into `HomeViewModel.handleDeepLink(url)`.

Output:
- `ShieldNotificationDispatcher` pure struct (source-shared into extension + main app) producing `UNNotificationRequest` with `userInfo["url"]: String`.
- `ShieldDeepLinkNotificationDelegate` NSObject delegate assigned at App launch, routing taps back through `homeModel.handleDeepLink`.
- Notification authorization requested lazily at App launch when status is `.notDetermined` (silent if denied — notifications are a best-effort fallback; the decision is still logged).
- 3 new XCTest cases: (1) dispatcher builds correct request, (2) delegate on tap → `HomeViewModel` routes to countdown when session active, (3) delegate on tap → HomeViewModel routes to home when no session.
- Extension import audit: `grep -c "^import UIKit$|^import SwiftUI$|^import FamilyControls$|^import Combine$" Extensions/ShieldActionExtension/ShieldActionExtension.swift` = 0.

    ```swift
    import Foundation
    import UserNotifications
    import os

    /// Pure builder + dispatcher for SHL-03 fallback. The extension uses
    /// `makeRequest(url:)` to build a `UNNotificationRequest` and `dispatch(url:log:)`
    /// to schedule it via `UNUserNotificationCenter.current().add(_:)`.
    ///
    /// Kept deliberately simple and dependency-free so it can be source-shared
    /// into `ShieldActionExtension` without inflating the 6 MB RAM ceiling
    /// (D-17). Imports: Foundation + UserNotifications + os only.
    ///
    /// Why local notification instead of direct URL open from extension:
    /// `ShieldActionDelegate` inherits from `NSObject` and exposes no
    /// `extensionContext`; the UIApplication runtime workaround is an App
    /// Store Guideline 2.5.1 rejection risk (see 04-03-SUMMARY.md
    /// §Post-Review Correction). Local notification is the canonical
    /// fallback documented in 04-RESEARCH.md Standard Stack §Alternatives.
    struct ShieldNotificationDispatcher {
        static let identifierPrefix = "com.kksw.DeluluDetox.shield-deeplink."
        static let userInfoURLKey = "url"
        static let notificationTitle = "DeluluDetox"
        static let notificationBody = "Dotknij, żeby wrócić do aplikacji."

        func makeRequest(url: URL) -> UNNotificationRequest {
            let content = UNMutableNotificationContent()
            content.title = Self.notificationTitle
            content.body = Self.notificationBody
            content.sound = nil
            content.userInfo = [Self.userInfoURLKey: url.absoluteString]
            let identifier = Self.identifierPrefix + UUID().uuidString
            return UNNotificationRequest(
                identifier: identifier,
                content: content,
                trigger: nil
            )
        }

        /// Best-effort schedule. Any UN* failure is logged and swallowed —
        /// the shield extension MUST NOT block `completionHandler(.close)` on
        /// notification success, per T-04-03-02 (shield process leak).
        func dispatch(url: URL, log: Logger) {
            let request = makeRequest(url: url)
            UNUserNotificationCenter.current().add(request) { error in
                if let error {
                    log.error("shield deeplink notif schedule failed: \\(String(describing: error), privacy: .public)")
                } else {
                    log.info("shield deeplink notif scheduled id=\\(request.identifier, privacy: .public)")
                }
            }
        }
    }
    ```

    ```swift
    import Foundation
    import UserNotifications
    import os

    /// Main-app notification center delegate responsible ONLY for shield-originated
    /// deep-link notifications. Receives banner taps, validates the `userInfo["url"]`
    /// payload, and forwards to `HomeViewModel.handleDeepLink(_:)` via an injected
    /// async handler. The handler is wired in `DeluluDetoxApp.init()` / `AppRootView`
    /// to reach `homeModel.handleDeepLink`.
    ///
    /// Safety: only acts on notifications whose identifier starts with
    /// `ShieldNotificationDispatcher.identifierPrefix` — other notification sources
    /// (future: Phase 6 NTF-01/NTF-02) are ignored here, letting the system pass
    /// them through to the default handler.
    @MainActor
    final class ShieldDeepLinkNotificationDelegate: NSObject, UNUserNotificationCenterDelegate {

        typealias DeepLinkHandler = @Sendable (URL) async -> Void

        private let handler: DeepLinkHandler
        private let log = Logger(
            subsystem: "com.kksw.DeluluDetox",
            category: "ShieldDeepLinkNotification"
        )

        init(handler: @escaping DeepLinkHandler) {
            self.handler = handler
            super.init()
        }

        // Show banner while app is in foreground — user may have shield up when
        // the notification fires; banner tap dismisses shield + foregrounds app.
        nonisolated func userNotificationCenter(
            _ center: UNUserNotificationCenter,
            willPresent notification: UNNotification,
            withCompletionHandler completionHandler: @escaping (UNNotificationPresentationOptions) -> Void
        ) {
            guard notification.request.identifier.hasPrefix(ShieldNotificationDispatcher.identifierPrefix) else {
                completionHandler([])
                return
            }
            completionHandler([.banner])
        }

        nonisolated func userNotificationCenter(
            _ center: UNUserNotificationCenter,
            didReceive response: UNNotificationResponse,
            withCompletionHandler completionHandler: @escaping () -> Void
        ) {
            let identifier = response.notification.request.identifier
            let userInfo = response.notification.request.content.userInfo

            guard identifier.hasPrefix(ShieldNotificationDispatcher.identifierPrefix) else {
                completionHandler()
                return
            }
            guard let urlString = userInfo[ShieldNotificationDispatcher.userInfoURLKey] as? String,
                  let url = URL(string: urlString),
                  url.scheme == "deluludetox" else {
                Task { @MainActor [log] in
                    log.error("shield deeplink notif tap: invalid userInfo")
                }
                completionHandler()
                return
            }

            Task { @MainActor [handler, log] in
                log.info("shield deeplink notif tap url=\\(url.absoluteString, privacy: .public)")
                await handler(url)
                completionHandler()
            }
        }
    }
    ```
